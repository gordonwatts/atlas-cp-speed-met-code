cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MetSoftTerm LANGUAGES CXX)
endif()

option(MET_SOFT_STOCK_CHECK "Build the release soft-term and Final comparison" OFF)

if(NOT TARGET correctionlib)
    find_package(correctionlib CONFIG REQUIRED)
endif()

add_library(met_soft STATIC soft.cxx provider.cxx)
add_library(atlas::met_soft ALIAS met_soft)
target_include_directories(met_soft PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(met_soft PUBLIC correctionlib)
target_compile_features(met_soft PUBLIC cxx_std_20)
set_target_properties(met_soft PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(met_soft PRIVATE -ffp-contract=off)
endif()

add_executable(met_soft_smoke smoke.cxx)
target_link_libraries(met_soft_smoke PRIVATE met_soft)
if(MET_SOFT_STOCK_CHECK)
    if(NOT TARGET AnalysisBase::METUtilitiesLib)
        find_package(AnalysisBase 25.2.117 EXACT CONFIG REQUIRED)
    endif()
    add_executable(met_soft_stock_compare stock_compare.cxx)
    target_compile_features(met_soft_stock_compare PRIVATE cxx_std_23)
    target_compile_definitions(met_soft_stock_compare PRIVATE XAOD_STANDALONE XAOD_ANALYSIS)
    target_link_libraries(met_soft_stock_compare PRIVATE met_soft
        AnalysisBase::ColumnarModeLib AnalysisBase::xAODRootAccess AnalysisBase::METUtilitiesLib)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(met_soft_stock_compare PRIVATE -ffp-contract=off)
    endif()
endif()
