cmake_minimum_required(VERSION 3.25)
project(AtlasMetColumnar LANGUAGES CXX)
find_package(correctionlib CONFIG REQUIRED)
add_subdirectory(../maker_adapter maker_adapter)
add_subdirectory(../soft soft EXCLUDE_FROM_ALL)
set(ATLAS_MET_SMALL_JVT "" CACHE PATH "Directory with the frozen small_jvt source")
set(ATLAS_MET_PINNED_SDK "" CACHE PATH "Pinned CorrectionLib checkout with dependencies")
set(ATLAS_MET_LWTNN "" CACHE FILEPATH "Pinned PIC lwtnn static library")
if(NOT EXISTS "${ATLAS_MET_SMALL_JVT}/small_jvt.cxx" OR NOT EXISTS "${ATLAS_MET_LWTNN}")
    message(FATAL_ERROR "Supply the reviewed small-R NNJVT source and pinned lwtnn library")
endif()
add_library(atlas_met_columnar SHARED transport.cpp "${ATLAS_MET_SMALL_JVT}/small_jvt.cxx")
target_compile_features(atlas_met_columnar PRIVATE cxx_std_20)
target_include_directories(atlas_met_columnar PRIVATE "${ATLAS_MET_SMALL_JVT}"
    "${ATLAS_MET_PINNED_SDK}/lwtnn/include" "${ATLAS_MET_PINNED_SDK}/eigen"
    "${ATLAS_MET_PINNED_SDK}/rapidjson/include")
target_link_libraries(atlas_met_columnar PRIVATE atlas_met_maker_correction atlas::met_soft "${ATLAS_MET_LWTNN}")
target_link_options(atlas_met_columnar PRIVATE -Wl,-z,defs)
option(ATLAS_MET_M27_DIAGNOSTICS "Compile focused jet track-count validation diagnostics" OFF)
if(ATLAS_MET_M27_DIAGNOSTICS)
    target_compile_definitions(atlas_met_columnar PRIVATE ATLAS_MET_M27_DIAGNOSTICS)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(atlas_met_columnar PRIVATE -fno-fast-math -ffp-contract=off)
endif()

option(ATLAS_MET_COLUMNAR_PREPARATION "Build the nominal preparation provider for a full runtime" OFF)
set(runtime_libraries atlas_met_columnar atlas_met_maker_correction atlas_met_maker_provider)
if(ATLAS_MET_COLUMNAR_PREPARATION)
    add_subdirectory(../preparation preparation EXCLUDE_FROM_ALL)
    add_dependencies(atlas_met_columnar met_preparation_provider)
    list(APPEND runtime_libraries met_preparation_provider)
endif()
set_target_properties(${runtime_libraries} PROPERTIES INSTALL_RPATH "$ORIGIN")
install(TARGETS ${runtime_libraries} LIBRARY DESTINATION native)
