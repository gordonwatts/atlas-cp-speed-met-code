cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(MetMakerAdapter LANGUAGES CXX)
endif()

if(NOT TARGET correctionlib)
    find_package(correctionlib CONFIG REQUIRED)
endif()
if(NOT TARGET atlas_met_maker)
    add_subdirectory(../maker maker)
endif()
if(NOT TARGET met_significance_nodes)
    add_subdirectory(../significance significance EXCLUDE_FROM_ALL)
endif()

add_library(atlas_met_maker_correction SHARED src/maker_correction.cpp src/significance_correction.cpp)
target_include_directories(atlas_met_maker_correction PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(atlas_met_maker_correction PUBLIC atlas::met_maker correctionlib met_significance_nodes)
target_compile_features(atlas_met_maker_correction PUBLIC cxx_std_20)
set_target_properties(atlas_met_maker_correction PROPERTIES INSTALL_RPATH "$ORIGIN")

add_library(atlas_met_maker_provider MODULE src/module.cpp)
target_link_libraries(atlas_met_maker_provider PRIVATE atlas_met_maker_correction)
set_target_properties(atlas_met_maker_provider PROPERTIES INSTALL_RPATH "$ORIGIN")

option(ATLAS_MET_ADAPTER_BUILD_SMOKE "Build the maker provider boundary smoke" OFF)
if(ATLAS_MET_ADAPTER_BUILD_SMOKE)
    add_executable(atlas_met_maker_adapter_smoke tests/smoke.cpp)
    target_link_libraries(atlas_met_maker_adapter_smoke PRIVATE atlas_met_maker_correction)
    target_include_directories(atlas_met_maker_adapter_smoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/src)
    add_executable(atlas_met_significance_event_smoke tests/significance_smoke.cpp)
    target_link_libraries(atlas_met_significance_event_smoke PRIVATE atlas_met_maker_correction)
endif()
