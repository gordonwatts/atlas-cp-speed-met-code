cmake_minimum_required(VERSION 3.16)
project(atlas_met_maker LANGUAGES CXX)

add_library(atlas_met_maker STATIC src/maker.cpp)
add_library(atlas::met_maker ALIAS atlas_met_maker)
target_include_directories(atlas_met_maker PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(atlas_met_maker PUBLIC cxx_std_20)
set_target_properties(atlas_met_maker PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(atlas_met_maker PRIVATE -fno-fast-math -ffp-contract=off)
endif()

option(ATLAS_MET_MAKER_BUILD_TESTS "Build the MET maker source-rule tests" OFF)
if(ATLAS_MET_MAKER_BUILD_TESTS)
    enable_testing()
    add_executable(atlas_met_maker_cases tests/maker_cases.cpp)
    target_link_libraries(atlas_met_maker_cases PRIVATE atlas::met_maker)
    add_test(NAME atlas_met_maker_cases COMMAND atlas_met_maker_cases)
endif()
